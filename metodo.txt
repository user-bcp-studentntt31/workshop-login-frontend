mcpgh() {
  local MCP_PROXY="/c/Users/Usuario/AppData/Roaming/npm/mcp-proxy.cmd"
  local NODE_WIN='C:\Progra~1\nodejs\node.exe'
  local MCP_SERVER_JS_WIN='C:\Users\Usuario\AppData\Roaming\npm\node_modules\mcp-github-project-manager\build\index.js'
  local PROJECT_DIR="/c/Users/Usuario/Documents/t53174-felix-chacaliaza-gutierrez/bcp-ai-lab-day-02"
  local ENV_FILE="$PROJECT_DIR/.env"

  if [[ ! -f "$MCP_PROXY" ]]; then
    echo "No existe archivo: $MCP_PROXY"
    return 1
  fi

  if [[ ! -f "$ENV_FILE" ]]; then
    echo "No existe $ENV_FILE"
    return 1
  fi

  while IFS= read -r line || [[ -n "$line" ]]; do
    line="${line%$'\r'}"
    [[ -z "$line" || "$line" =~ ^[[:space:]]*# ]] && continue
    export "$line"
  done < "$ENV_FILE"

  "$MCP_PROXY" --host 127.0.0.1 --port 8081 --server stream -- \
  "$NODE_WIN" "$MCP_SERVER_JS_WIN"
}
